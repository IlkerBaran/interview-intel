{# ───────────────────────────────────────────── Reusable Jinja2 macros for Interview Intel. Import in any template with: {% from "macros.html" import category_badge, urgency_badge, form_field %} ───────────────────────────────────────────────── #} {% macro category_badge(category) %} {% if category == 'interview_invitation' %} Interview Invitation {% elif category == 'offer' %} Offer {% elif category == 'rejection' %} Rejection {% elif category == 'scheduling' %} Scheduling {% elif category == 'recruiter_outreach' %} Recruiter Outreach {% elif category == 'follow_up' %} Follow-up {% else %} {{ category | replace('_', ' ') | title }} {% endif %} {% endmacro %} {% macro urgency_badge(urgency) %} {% if urgency == 'high' %} High {% elif urgency == 'medium' %} Medium {% elif urgency == 'low' %} Low {% else %} {{ urgency | title }} {% endif %} {% endmacro %} {% macro form_field(field, optional=false, autofocus=false) %} {# One labelled field + its errors. The "optional" marker is nested INSIDE the label so the .form-label .optional rule applies. #}
{{ field(class="form-input" + (" input-error" if field.errors else ""), autofocus=autofocus) }} {% if field.errors %}
{% for error in field.errors %} {{ error }} {% endfor %}
{% endif %}
{% endmacro %} {# ───────────────────────────────────────────── Timestamp rendering. Stored timestamps are UTC. This emits the instant in a machine-readable form plus a server-rendered UTC fallback; static/js/localtime.js rewrites the text into the viewer's own timezone on load. The fallback is explicitly suffixed "UTC" on the time-bearing presets so a JS-disabled client reads a correct labelled time rather than an unlabelled wrong one. Do NOT use this for user-entered date-only values such as JobApplication.applied_date (db.Date) — those never passed through UTC and must not be shifted. presets: datetime | datetime_short | date | date_short | month_day_year ───────────────────────────────────────────────── #} {% macro localtime(dt, preset="date") %} {%- if dt -%} {%- if preset == "datetime" -%} {%- set fallback = dt.strftime('%B %d, %Y at %I:%M %p') ~ ' UTC' -%} {%- elif preset == "datetime_short" -%} {%- set fallback = dt.strftime('%b %d, %Y %H:%M') ~ ' UTC' -%} {%- elif preset == "date_short" -%} {%- set fallback = dt.strftime('%b %d') -%} {%- elif preset == "month_day_year" -%} {%- set fallback = dt.strftime('%B') ~ ' ' ~ dt.day ~ ', ' ~ dt.strftime('%Y') -%} {%- else -%} {%- set fallback = dt.strftime('%b %d, %Y') -%} {%- endif -%} {{ fallback }} {%- endif -%} {% endmacro %}
